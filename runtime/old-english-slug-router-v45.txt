     9	function mdo_en_is_request(): bool {
    10	    if ( empty( $_SERVER['REQUEST_URI'] ) ) { return false; }
    11	    $path = wp_parse_url( wp_unslash( $_SERVER['REQUEST_URI'] ), PHP_URL_PATH );
    12	    return is_string( $path ) && ( $path === '/en' || strpos( $path, '/en/' ) === 0 );
    13	}
    14	
    15	function mdo_en_segments(): array {
    16	    if ( ! mdo_en_is_request() ) { return array(); }
    17	    $path = (string) wp_parse_url( wp_unslash( $_SERVER['REQUEST_URI'] ), PHP_URL_PATH );
    18	    $relative = trim( preg_replace( '#^/en/?#', '', $path ), '/' );
    19	    if ( $relative === '' ) { return array(); }
    20	    return array_values( array_filter( array_map( 'sanitize_title', explode( '/', $relative ) ), 'strlen' ) );
    21	}

    27	function mdo_en_slug_for_post( int $post_id ): string {
    28	    if ( (string) get_post_meta( $post_id, '_en_US_published', true ) !== '1' ) { return ''; }
    29	    return sanitize_title( wp_strip_all_tags( (string) get_post_meta( $post_id, '_en_US_post_name', true ) ) );
    30	}

    32	function mdo_en_find_by_slug( string $post_type, string $slug ): int {
    33	    $slug = sanitize_title( $slug );
    34	    if ( $slug === '' ) { return 0; }
    35	    $ids = get_posts( array(
    36	        'post_type'        => $post_type,
    37	        'post_status'      => 'publish',
    38	        'posts_per_page'   => 2,
    39	        'fields'           => 'ids',
    40	        'no_found_rows'    => true,
    41	        'suppress_filters' => true,
    42	        'meta_query'       => array(
    43	            'relation' => 'AND',
    44	            array( 'key' => '_en_US_post_name', 'value' => $slug, 'compare' => '=' ),
    45	            array( 'key' => '_en_US_published', 'value' => '1', 'compare' => '=' ),
    46	        ),
    47	    ) );
    48	    return count( $ids ) === 1 ? (int) $ids[0] : 0;
    49	}

    71	function mdo_en_translate_internal_url( string $url ): string {
    72	    if ( $url === '' || strpos( $url, '#' ) === 0 || stripos( $url, 'mailto:' ) === 0 || stripos( $url, 'tel:' ) === 0 || stripos( $url, 'javascript:' ) === 0 ) { return $url; }
    73	    $root = mdo_en_root();
    74	    $root_host = wp_parse_url( $root, PHP_URL_HOST );
    75	    $host = wp_parse_url( $url, PHP_URL_HOST );
    76	    if ( $host && $root_host && strtolower( (string) $host ) !== strtolower( (string) $root_host ) ) { return $url; }
    77	
    78	    $query = wp_parse_url( $url, PHP_URL_QUERY );
    79	    $fragment = wp_parse_url( $url, PHP_URL_FRAGMENT );
    80	    $suffix = '';
    81	    if ( is_string( $query ) && $query !== '' ) { $suffix .= '?' . $query; }
    82	    if ( is_string( $fragment ) && $fragment !== '' ) { $suffix .= '#' . $fragment; }
    83	
    84	    $path = (string) wp_parse_url( $url, PHP_URL_PATH );
    85	    $parts = array_values( array_filter( explode( '/', trim( $path, '/' ) ), 'strlen' ) );
    86	    if ( ! $parts ) { return $root . '/en/' . $suffix; }
    87	
    88	    $is_en = strtolower( $parts[0] ) === 'en';
    89	    if ( $is_en ) { array_shift( $parts ); }
    90	    if ( ! $parts ) { return $root . '/en/' . $suffix; }
    91	
    92	    if ( count( $parts ) === 1 ) {
    93	        $english_id = mdo_en_find_by_slug( 'page', $parts[0] );
    94	        if ( $english_id ) { return mdo_en_page_url( $english_id ) . $suffix; }
    95	        $native = mdo_en_native_page_from_slug( $parts[0] );
    96	        if ( $native ) {
    97	            $translated = mdo_en_page_url( (int) $native->ID );
    98	            if ( $translated !== '' ) { return $translated . $suffix; }
    99	        }
   100	    }
   101	
   102	    if ( count( $parts ) === 2 && in_array( sanitize_title( $parts[0] ), array( 'producto', 'product' ), true ) ) {
   103	        $english_id = mdo_en_find_by_slug( 'product', $parts[1] );
   104	        if ( $english_id ) { return mdo_en_product_url( $english_id ) . $suffix; }
   105	        $native = mdo_en_native_product_from_slug( $parts[1] );
   106	        if ( $native ) {
   107	            $translated = mdo_en_product_url( (int) $native->ID );
   108	            if ( $translated !== '' ) { return $translated . $suffix; }
   109	        }
   110	    }
   111	
   112	    return $url;
   113	}

   115	add_action( 'parse_request', static function ( WP $wp ): void {
   116	    $parts = mdo_en_segments();
   117	    if ( ! $parts ) { return; }
   118	
   119	    if ( count( $parts ) === 2 && $parts[0] === 'product' ) {
   120	        $id = mdo_en_find_by_slug( 'product', $parts[1] );
   121	        if ( ! $id ) { return; }
   122	        foreach ( array( 'error', 'name', 'pagename', 'page_id', 'attachment', 'product', 'product_cat', 'product_tag' ) as $key ) { unset( $wp->query_vars[ $key ] ); }
   123	        $wp->query_vars['post_type'] = 'product';
   124	        $wp->query_vars['p'] = $id;
   125	        return;
   126	    }
   127	
   128	    if ( count( $parts ) !== 1 ) { return; }
   129	    $id = mdo_en_find_by_slug( 'page', $parts[0] );
   130	    if ( ! $id ) { return; }
   131	    foreach ( array( 'error', 'name', 'pagename', 'page_id', 'attachment', 'p' ) as $key ) { unset( $wp->query_vars[ $key ] ); }
   132	    if ( $id === (int) get_option( 'woocommerce_shop_page_id' ) ) {
   133	        $wp->query_vars['post_type'] = 'product';
   134	    } else {
   135	        unset( $wp->query_vars['post_type'] );
   136	        $wp->query_vars['page_id'] = $id;
   137	    }
   138	}, PHP_INT_MAX );

   165	    }

   222	function mdo_en_find_term_by_slug( string $taxonomy, string $slug ) {
   223	    if ( ! taxonomy_exists( $taxonomy ) ) { return null; }
   224	    $slug = sanitize_title( $slug );
   225	    if ( $slug === '' ) { return null; }
   226	    $terms = get_terms( array(
   227	        'taxonomy'   => $taxonomy,
   228	        'hide_empty' => false,
   229	        'number'     => 2,
   230	        'meta_query' => array(
   231	            'relation' => 'AND',
   232	            array( 'key' => '_en_US_slug', 'value' => $slug, 'compare' => '=' ),
   233	            array( 'key' => '_en_US_published', 'value' => '1', 'compare' => '=' ),
   234	        ),
   235	    ) );
   236	    if ( is_wp_error( $terms ) || count( $terms ) !== 1 ) { return null; }
   237	    return $terms[0];
   238	}

   271	add_action( 'parse_request', static function ( WP $wp ): void {
   272	    if ( ! mdo_en_is_request() ) { return; }
   273	    $parts = mdo_en_segments();
   274	    if ( count( $parts ) !== 2 ) { return; }
   275	
   276	    if ( $parts[0] === 'store' ) {
   277	        foreach ( array( 'error', 'name', 'pagename', 'page_id', 'p', 'attachment', 'product', 'product_cat', 'product_tag', 'category_name' ) as $key ) { unset( $wp->query_vars[ $key ] ); }

   280	        return;
   281	    }
   282	
   283	    $taxonomy = mdo_en_taxonomy_for_base( $parts[0] );
   284	    if ( $taxonomy === '' ) { return; }
   285	    $term = mdo_en_find_term_by_slug( $taxonomy, $parts[1] );
   286	    if ( ! $term instanceof WP_Term ) { return; }
   287	
   288	    foreach ( array( 'error', 'name', 'pagename', 'page_id', 'p', 'attachment', 'product', 'product_cat', 'product_tag', 'category_name', 'cat', 'tag' ) as $key ) { unset( $wp->query_vars[ $key ] ); }
   289	    if ( $taxonomy === 'product_cat' ) {
   290	        $wp->query_vars['post_type'] = 'product';
   291	        $wp->query_vars['product_cat'] = $term->slug;
   292	    } elseif ( $taxonomy === 'product_tag' ) {
   293	        $wp->query_vars['post_type'] = 'product';
   294	        $wp->query_vars['product_tag'] = $term->slug;
   295	    } else {
   296	        unset( $wp->query_vars['post_type'] );
   297	        $wp->query_vars['category_name'] = $term->slug;
   298	    }
   299	}, PHP_INT_MAX );

   373	/*
   374	 * Base router sets post_type=product so its custom catalogue loader stays
   375	 * active. Product taxonomy aliases must instead mirror native WC taxonomy
   376	 * semantics; the taxonomy query var alone is enough and prevents is_shop().
   377	 */
   378	add_action( 'parse_request', static function ( WP $wp ): void {
   379	    if ( ! mdo_en_is_request() ) { return; }
   380	    $parts = mdo_en_segments();
   381	    if ( count( $parts ) === 2 ) {
   382	        $taxonomy = mdo_en_taxonomy_for_base( $parts[0] );
   383	        if ( in_array( $taxonomy, array( 'product_cat', 'product_tag' ), true ) ) {
   384	            $term = mdo_en_find_term_by_slug( $taxonomy, $parts[1] );
   385	            if ( $term instanceof WP_Term ) {
   386	                unset( $wp->query_vars['post_type'] );
   387	            }
   388	        }
   389	        return;
   390	    }
   391	
   392	    if ( count( $parts ) !== 1 ) { return; }
   393	    $post_id = mdo_en_find_by_slug( 'post', $parts[0] );
   394	    if ( ! $post_id ) { return; }
   395	    foreach ( array( 'error', 'name', 'pagename', 'page_id', 'attachment', 'product', 'product_cat', 'product_tag', 'category_name' ) as $key ) {
   396	        unset( $wp->query_vars[ $key ] );
   397	    }
   398	    $wp->query_vars['post_type'] = 'post';
   399	    $wp->query_vars['p'] = $post_id;
   400	}, PHP_INT_MAX );
