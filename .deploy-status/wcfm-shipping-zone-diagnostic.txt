	function wcfmmp_get_pickup_address( $vendor_id ) {
		if( !wcfm_is_vendor($vendor_id) ) {
			$pickup_address = '';
		} else {
			$store_user    = wcfmmp_get_store( $vendor_id );
			$pickup_address = $store_user->get_address_string();	
		}
		if ($pickup_address && apply_filters('wcfm_is_pref_multi_store', true)) {
			global $WCFMu;
			$only_address = true;
			$pickup_address = $WCFMu->wcfmu_multi_store->wcfmu_return_closest_pickup_address($pickup_address, $vendor_id, $only_address);
		}
		return apply_filters('wcfmmp_store_pickup_address', $pickup_address);
	}
}
